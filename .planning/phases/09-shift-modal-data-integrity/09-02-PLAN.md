       const formatCommentHeader = (comment: Comment, currentUser: any) => {
         // Format: [Name] • [Role Tag] • [DD.MM HH:mm]

         const date = new Date(comment.created_at);
         const dayMonth = date.toLocaleDateString('ru-RU', {
           day: '2-digit',
           month: '2-digit'
         }); // "14.02"
         const time = date.toLocaleTimeString('ru-RU', {
           hour: '2-digit',
           minute: '2-digit'
         }); // "12:30"

         // Role tag mapping
         const roleLabels: Record<string, string> = {
           'admin': 'АДМИН',
           'driver': 'ВОДИТЕЛЬ',
           'foreman': 'ПРОРАБ'
         };
         const roleTag = comment.author_role
           ? roleLabels[comment.author_role] || comment.author_role.toUpperCase()
           : null;

         return {
           name: comment.author,
           roleTag,
           timestamp: `${dayMonth} ${time}`
         };
       };
